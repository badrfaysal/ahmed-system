            ->where('installment_months', 0)->where('category', 'خدمات')->sum('profit');

        $profitGas = (float) DB::table('installments')->where('category', 'بنزينة')->sum('profit');

        $profitAssetSales = (float) (DB::table('assets')->where('status', 'sold')
            ->whereRaw('selling_price > current_value')
            ->selectRaw('COALESCE(SUM(selling_price - current_value), 0) as profit')->value('profit') ?? 0);

        $additionalIncomes = (float) DB::table('financial_transactions')
            ->where('type', 'income')->where('status', 'active')
            ->where('notes', 'like', '%أرباح فرق سعر%')->sum('amount');

        $totalGrossRevenue = $profitInstallments + $profitInventory + $profitDirectProducts
            + $profitServices + $profitGas + $profitAssetSales + $additionalIncomes;

        $expensesGeneral = (float) DB::table('financial_transactions')->where('type', 'general_expense')
            ->where('status', 'active')->where('notes', 'not like', '%إهلاك أصل ثابت%')
            ->where('notes', 'not like', '%عمولة تلقائية%')->where('notes', 'not like', '%إعدام ديون%')->sum('amount');

        $totalCommissions = (float) DB::table('financial_transactions')->where('type', 'general_expense')
            ->where('status', 'active')->where('notes', 'like', '%عمولة تلقائية%')->sum('amount');

        $lossesDepreciation = (float) DB::table('financial_transactions')->where('type', 'general_expense')
            ->where('status', 'active')->where('notes', 'like', '%إهلاك أصل ثابت%')->sum('amount');

        $lossesBadDebts = (float) DB::table('financial_transactions')->where('type', 'general_expense')
            ->where('status', 'active')->where('notes', 'like', '%إعدام ديون%')->sum('amount');

        $lossesReturns = (float) (DB::table('sale_returns')->sum('loss_amount') ?? 0);
        $lossesDiscounts = (float) DB::table('financial_transactions')->where('type', 'discount')
            ->where('status', 'active')->sum('amount');
        $lossesAssetSales = (float) (DB::table('assets')->where('status', 'sold')
            ->whereRaw('selling_price < current_value')
            ->selectRaw('COALESCE(SUM(current_value - selling_price), 0) as loss')->value('loss') ?? 0);
        $expensesSalaries = (float) DB::table('financial_transactions')->where('type', 'salary_expense')
            ->where('status', 'active')->sum('amount');

        $totalDeductions = $expensesGeneral + $expensesSalaries + $totalCommissions + $lossesDepreciation
            + $lossesReturns + $lossesDiscounts + $lossesBadDebts + $lossesAssetSales;

        $netBookProfit = $totalGrossRevenue - $totalDeductions;

        $activeInstallments = DB::table('installments')->where('remaining_balance', '>', 0)->get(['profit', 'remaining_balance', 'total_after_interest', 'down_payment']);
        $uncollectedProfit = $activeInstallments->sum(fn ($inst) => self::uncollectedProfitShare($inst));

        $realCollectedProfit = max(0, $netBookProfit - $uncollectedProfit);
        $totalDistributedProfits = (float) DB::table('financial_transactions')->where('type', 'profit_distribution')->sum('amount');
        $remainingCompanyProfit = max(0, $realCollectedProfit - $totalDistributedProfits);

        return compact(
            'liquidity', 'projectsValue', 'inventoryAssets', 'fixedAssets',
            'installmentsSystemDebts', 'otherDebtsForUs', 'totalDebtsForUs', 'totalDebtsOnUs', 'capital',
            'profitInstallments', 'profitInventory', 'profitDirectProducts', 'profitServices', 'profitGas',
            'profitAssetSales', 'additionalIncomes', 'totalGrossRevenue',
            'expensesGeneral', 'totalCommissions', 'lossesDepreciation', 'lossesBadDebts',
            'lossesReturns', 'lossesDiscounts', 'lossesAssetSales', 'expensesSalaries', 'totalDeductions',
            'netBookProfit', 'uncollectedProfit', 'realCollectedProfit', 'totalDistributedProfits', 'remainingCompanyProfit'
        );
    }

    /** مبلغ يُسترد عند إلغاء دفعة (كاش + خصم مُطبَّق في نفس العملية) */
    public static function paymentReversalAmount(object $payment): float
    {
        $cash = (float) ($payment->amount_paid ?? 0);
        $disc = 0;
        if (Schema::hasColumn('installment_payments', 'discount_applied')) {
            $disc = (float) ($payment->discount_applied ?? 0);
        }
        return $cash + $disc;
    }

    public static function restoreInstallmentAfterPaymentReversal(int $installmentId, object $payment): void
    {
        $restore = self::paymentReversalAmount($payment);
        if ($restore <= 0) {
            return;
        }
        DB::table('installments')->where('id', $installmentId)->increment('remaining_balance', $restore);
        $disc = Schema::hasColumn('installment_payments', 'discount_applied')
            ? (float) ($payment->discount_applied ?? 0) : 0;
        if ($disc > 0) {
            DB::table('installments')->where('id', $installmentId)->increment('profit', $disc);
        }
        if (Schema::hasColumn('installments', 'close_reason')) {
            DB::table('installments')->where('id', $installmentId)->update(['close_reason' => null]);
        }
